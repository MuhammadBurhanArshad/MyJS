JavaScript String Methods

List of String Methods

MethodDescription
lengthCounts characters
toLowerCase()Converts to lowercase
toUpperCase()Converts to uppercase
includes()Checks if value exists
startsWith()Checks start of string
endsWith()Checks end of string
search()Position of match
match()Matches in array (regex)
indexOf()First index
lastIndexOf()Last index
replace()Replaces a word
trim()Removes extra spaces
split()Splits into array
concat()Joins strings
repeat()Repeats string
slice()Extracts portion
substring()Extracts between indexes
charAt()Character at position
charCodeAt()Unicode of character
toString()Convert to string
valueOf()Return primitive value

1. length

var str = "JavaScript is a Great Language";
str.length;   // 30

2. toLowerCase / toUpperCase

"Burhan".toLowerCase();   // "burhan"
"Burhan".toUpperCase();   // "BURHAN"

3. includes()

"Burhan".includes("Burhan");   // true
// Case-sensitive

4. startsWith / endsWith

var str = "Burhan is a good boy";

str.startsWith("Burhan");   // true
str.endsWith("boy");        // true

5. search()

var str = "Burhan Arshad";
str.search("Arshad");   // 8
str.search("Mango");    // -1

6. match()

var str = "Burhan is a good boy";
str.match(/is/g);
// ["is"]  (uses regular expressions)

7. indexOf / lastIndexOf

var str = "Burhan is a good boy";
str.indexOf("is");       // 7
str.lastIndexOf("is");   // 7

8. replace()

var str = "JavaScript is a great language";
str.replace("JavaScript", "PHP");
// "PHP is a great language"

9. trim()

var str = "   Burhan   ";
str.trim();
// "Burhan"  (removes leading and trailing spaces)
