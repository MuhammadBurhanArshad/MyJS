JavaScript Functions with Return Value

The return keyword sends a value back to the caller.

Syntax

function functionName(parameter1, parameter2) {
    statements
    return value;
}

var a = functionName(argument1, argument2);

Example — Sum with Return

function sum(a, b) {
    return a + b;
}

var result = sum(10, 20);   // 30
document.write(result);

More Examples

Multiply

function multiply(x, y) {
    return x * y;
}

var total = multiply(5, 4);   // 20

Greet

function greet(name) {
    return "Hello, " + name + "!";
}

var msg = greet("Burhan");   // Hello, Burhan!

Max of Two Numbers

function max(a, b) {
    if (a > b) {
        return a;
    } else {
        return b;
    }
}

document.write(max(15, 25));   // 25
